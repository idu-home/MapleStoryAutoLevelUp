### 1.2 模板匹配核心算法

```python
def find_pattern_sqdiff(img, img_pattern, last_result=None, mask=None, 
                       local_search_radius=50, global_threshold=0.4):
    # 局部搜索优化
    if last_result is not None and global_threshold > 0.0:
        # 在上次结果附近搜索，提高效率
        lx, ly = last_result
        x0 = max(0, lx - local_search_radius)
        y0 = max(0, ly - local_search_radius)
        # ... 局部搜索逻辑
    
    # 全局搜索回退
    res = cv2.matchTemplate(img, img_pattern, cv2.TM_SQDIFF_NORMED, mask=mask)
    min_val, max_val, min_loc, max_loc = cv2.minMaxLoc(res)
    
    return min_loc, min_val, False
```

**算法特点**:
- **SQDIFF_NORMED**: 使用归一化平方差匹配方法
- **局部搜索优化**: 基于上次结果进行局部搜索，提高效率
- **掩码支持**: 支持透明背景处理
- **阈值控制**: 可配置的匹配阈值

```python
def get_player_location_by_party_red_bar(self):
    # 转换为HSV颜色空间
    img_hsv = cv2.cvtColor(img_camera, cv2.COLOR_BGR2HSV)
    lower_red = to_opencv_hsv(self.cfg["party_red_bar"]["lower_red"])
    upper_red = to_opencv_hsv(self.cfg["party_red_bar"]["upper_red"])
    mask_red = cv2.inRange(img_hsv, lower_red, upper_red)
    
    # 轮廓检测
    contours, _ = cv2.findContours(mask_red, cv2.RETR_EXTERNAL, 
                                   cv2.CHAIN_APPROX_SIMPLE)
    
    # 过滤轮廓
    for c in contours:
        x, y, w, h = cv2.boundingRect(c)
        area = cv2.contourArea(c)
        fill_rate = float(area) / (h*w)
        
        # 红色血条的特征：细长、高填充率
        if 5 <= h <= 7 and 1 <= w <= 50 and 10 <= area and fill_rate >= 0.7:
            # 找到红色血条

            loc_player = (x + self.cfg["party_red_bar"]["offset"][0],
                         y + self.cfg["party_red_bar"]["offset"][1])
            return loc_player, loc_party_red_bar
```

#### 2.3.1 多种检测模式

```python
def get_monsters_in_range(self, top_left, bottom_right):
    for monster_name, monster_imgs in self.monsters_info.items():
        for img_monster, mask_monster in monster_imgs:
            if self.cfg["monster_detect"]["mode"] == "template_free":
                # 无模板模式：检测黑色像素
                black_mask = np.all(img_roi == [0, 0, 0], axis=2).astype(np.uint8) * 255
                # 排除玩家角色区域
                black_mask[char_y_min:char_y_max, char_x_min:char_x_max] = 0
                # 形态学操作
                kernel = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (20, 20))
                closed_mask = cv2.morphologyEx(black_mask, cv2.MORPH_CLOSE, kernel)
                # 连通组件分析
                num_labels, labels, stats, centroids = \
                    cv2.connectedComponentsWithStats(closed_mask, connectivity=8)
                
            elif self.cfg["monster_detect"]["mode"] == "contour_only":
                # 轮廓模式：只使用黑色轮廓
                mask_pattern = np.all(img_monster == [0, 0, 0], axis=2).astype(np.uint8) * 255
                mask_roi = np.all(img_roi == [0, 0, 0], axis=2).astype(np.uint8) * 255
                # 高斯模糊
                blur = self.cfg["monster_detect"]["contour_blur"]
                img_monster_blur = cv2.GaussianBlur(mask_pattern, (blur, blur), 0)
                img_roi_blur = cv2.GaussianBlur(mask_roi, (blur, blur), 0)
                # 模板匹配
                res = cv2.matchTemplate(img_roi_blur, img_monster_blur, cv2.TM_SQDIFF_NORMED)
                
            elif self.cfg["monster_detect"]["mode"] == "grayscale":
                # 灰度模式
                img_monster_gray = cv2.cvtColor(img_monster, cv2.COLOR_BGR2GRAY)
                img_roi_gray = cv2.cvtColor(img_roi, cv2.COLOR_BGR2GRAY)
                res = cv2.matchTemplate(img_roi_gray, img_monster_gray, 
                                       cv2.TM_SQDIFF_NORMED, mask=mask_monster)
                
            elif self.cfg["monster_detect"]["mode"] == "color":
                # 彩色模式：最准确但最慢
                res = cv2.matchTemplate(img_roi, img_monster, 
                                       cv2.TM_SQDIFF_NORMED, mask=mask_monster)
```

**检测模式对比**:

| 模式 | 准确性 | 速度 | 适用场景 |
|------|--------|------|----------|
| `template_free` | 低 | 最快 | 简单场景，快速检测 |
| `contour_only` | 中 | 快 | 平衡性能和准确性 |
| `grayscale` | 中高 | 中 | 一般场景 |
| `color` | 最高 | 最慢 | 复杂场景，高精度要求 |

#### 2.3.3 非极大值抑制 (NMS)

```python
def nms(monsters, iou_threshold=0.3):
    boxes = []
    for m in monsters:
        x, y = m["position"]
        w, h = m["size"]
        # [x1, y1, x2, y2, score, original_data]
        boxes.append([x, y, x + w, y + h, m["score"], m])
    
    # 按分数排序
    boxes.sort(key=lambda x: x[4], reverse=True)
    
    keep = []
    while boxes:
        best = boxes.pop(0)
        keep.append(best[5])  # 保留最佳匹配
        
        # 移除重叠的检测框
        boxes = [b for b in boxes if get_iou(best, b) < iou_threshold]
    
    return keep
```

**NMS作用**:
- **去除重复**: 避免同一怪物被多次检测
- **保留最佳**: 保留置信度最高的检测结果
- **IoU阈值**: 可配置的重叠阈值

```python
def update_rune_location(self, img, img_debug, loc_player):
    # 计算检测范围
    h_rune_box = self.cfg["rune_detect"]["box_height"]
    w_rune_box = self.cfg["rune_detect"]["box_width"]
    x0 = max(0, loc_player[0] - w_rune_box // 2)
    y0 = max(0, loc_player[1] - h_rune_box)
    x1 = min(w, loc_player[0] + w_rune_box // 2)
    y1 = min(h, loc_player[1])
    
    # 分别匹配符文的各个部分
    matches = []
    for i, img_rune in enumerate(self.img_runes):
        mask = get_mask(img_rune, (0, 255, 0))
        loc, score, _ = find_pattern_sqdiff(img[y0:y1, x0:x1], img_rune, mask=mask)
        matches.append((i, loc, score, img_rune.shape))
    
    # 过滤好的匹配
    good_matches = [
        (i, loc, score, shape)
        for (i, loc, score, shape) in matches
        if score < self.cfg["rune_detect"]["diff_thres"]
    ]
    
    # 要求至少2个符文部分
    if len(good_matches) < 2:
        return
    
    # 验证符文几何关系
    x_centers = [x0 + loc[0] + shape[1] // 2 for (_, loc, _, shape) in good_matches]
    if max(x_centers) - min(x_centers) > 10:
        return  # 水平距离过大
    
    ys = [y0 + loc[1] for (_, loc, _, _) in good_matches]
    if not all(ys[i] < ys[i + 1] for i in range(len(ys) - 1)):
        return  # Y坐标不是严格递增
```

```python
def get_all_other_player_locations_on_minimap(img_minimap, red_bgr=(0, 0, 255)):
    # 智能选择容错范围
    tolerances = [10, 20, 30, 40]
```
